## **Overview**

**Goal**: Migrate to 64-bit database format while preserving `.root` extension and legacy hash tables.

**Strategy**: 
- **Version 6**: Legacy 32-bit format (current)
- **Version 7**: 64-bit format with legacy hash tables (new)
- **Version 8**: 64-bit format with modern hash tables (future)

```c
// 64-bit database record structure (Version 7)
typedef struct tydatabaserecord_64 {
    unsigned char systemid;        // 1 byte
    unsigned char versionnumber;   // 1 byte
    dbaddress availlist;          // 8 bytes (long long)
    short oldfnumdatabase;        // 2 bytes
    short flags;                  // 2 bytes
    dbaddress views[ctviews];     // 24 bytes (3 * 8 bytes)
    Handle releasestack;          // 8 bytes (void*)
    long fnumdatabase;            // 8 bytes (long long)
    long headerLength;            // 8 bytes (long long)
    short longversionMajor;       // 2 bytes
    short longversionMinor;       // 2 bytes
    union {
        char growthspace[22];     // 22 bytes (reduced for 64-bit alignment)
        struct {
            dbaddress availlistblock;  // 8 bytes
            handlestream availlistshadow; // 8 bytes
            boolean flreadonly;         // 1 byte
        } extensions;
    } u;
} tydatabaserecord_64;
```

### **Step 7: Add Migration Functions**

**File**: `Common/source/db.c`

**Add migration from 32-bit to 64-bit:**
```c
boolean migrate_32bit_to_64bit(hdldatabaserecord hdb) {
    // Create backup
    if (!create_root_backup(current_db_path)) {
        return false;
    }
    
    // Convert database structure
    tydatabaserecord_64 new_header;
    convert_32bit_header_to_64bit(hdb, &new_header);
    
    // Update version number
    new_header.versionnumber = 7;
    
    // Write new format to disk
    return write_64bit_database(current_db_path, &new_header);
}

boolean convert_32bit_header_to_64bit(hdldatabaserecord hdb, tydatabaserecord_64 *new_header) {
    // Copy basic fields
    new_header->systemid = (**hdb).systemid;
    new_header->versionnumber = 7;  // New version
    new_header->availlist = (dbaddress)(**hdb).availlist;  // 32-bit → 64-bit
    new_header->oldfnumdatabase = (**hdb).oldfnumdatabase;
    new_header->flags = (**hdb).flags;
    
    // Convert views array
    for (int i = 0; i < ctviews; i++) {
        new_header->views[i] = (dbaddress)(**hdb).views[i];  // 32-bit → 64-bit
    }
    
    new_header->releasestack = (**hdb).releasestack;
    new_header->fnumdatabase = (long)(**hdb).fnumdatabase;  // 32-bit → 64-bit
    new_header->headerLength = (long)(**hdb).headerLength;  // 32-bit → 64-bit
    new_header->longversionMajor = (**hdb).longversionMajor;
    new_header->longversionMinor = (**hdb).longversionMinor;
    
    return true;
}
```

### **Step 8: Add Backup Creation**

**File**: `Common/source/db.c`

**Add backup function:**
```c
boolean create_root_backup(const char* original_path) {
    char backup_path[PATH_MAX];
    snprintf(backup_path, sizeof(backup_path), "%s.rbk", original_path);
    
    // Copy file to backup
    FILE *src = fopen(original_path, "rb");
    FILE *dst = fopen(backup_path, "wb");
    
    if (!src || !dst) {
        if (src) fclose(src);
        if (dst) fclose(dst);
        return false;
    }
    
    // Copy file contents
    char buffer[4096];
    size_t bytes;
    while ((bytes = fread(buffer, 1, sizeof(buffer), src)) > 0) {
        fwrite(buffer, 1, bytes, dst);
    }
    
    fclose(src);
    fclose(dst);
    return true;
}
```
